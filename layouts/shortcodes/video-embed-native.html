{{- $video := .Get "video_url" -}} {{- /* Our own screencasts are narrated, and the recorder publishes the narration as WebVTT next to each video (same name, .vtt). Fetch it at build time and render it as a transcript: the voice-over becomes readable for anyone who cannot play sound, and indexable for search. A would need CORS on the asset bucket; a build-time fetch does not. A missing file (the video predates narration, or has not been re-recorded yet) renders nothing. */ -}} {{- $transcriptURL := .Get "transcript_url" -}} {{- if and (not $transcriptURL) (hasPrefix $video "https://marketing-assets.sbomify.com/screencasts/") -}} {{- $transcriptURL = replaceRE `\.webm$` ".vtt" $video -}} {{- end -}} {{- $transcript := slice -}} {{- with $transcriptURL -}} {{- with try (resources.GetRemote .) -}} {{- with .Err -}} {{- warnf "video-embed-native: transcript %s: %s" $transcriptURL . -}} {{- else with .Value -}} {{- range split (replace .Content "\r\n" "\n") "\n\n" -}} {{- $lines := split (trim . "\n") "\n" -}} {{- $cue := slice -}} {{- $text := false -}} {{- range $lines -}} {{- if $text -}}{{- $cue = $cue | append . -}}{{- end -}} {{- if in . "-->" -}}{{- $text = true -}}{{- end -}} {{- end -}} {{- with $cue -}}{{- $transcript = $transcript | append (delimit . " ") -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
